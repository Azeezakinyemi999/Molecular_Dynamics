def generate_orchestrator_sh(
    orch_job_name,
    orch_partition,
    orch_cpus_per_task,
    orch_mem,
    orch_time,
    orch_openmpi_ver,
    orch_cuda_version,
    orch_conda_env,
    orch_ld_paths,
    work_dir,
    out_py,
    out_sh: str,
) -> str:
    """Write diffusivity_run.sh and chmod +x it. Returns the output path."""

    orch_time_line = f'#SBATCH --time={orch_time}' if orch_time else ''
    _ld_lines = ('\n        ').join(
        f'export LD_LIBRARY_PATH={p}:$LD_LIBRARY_PATH'
        for p in orch_ld_paths
    )
    orch_output = f'{orch_job_name}_%j.out'

    content = textwrap.dedent(f"""\
        #!/bin/bash
        #SBATCH --job-name={orch_job_name}
        #SBATCH --partition={orch_partition}
        #SBATCH --ntasks=1
        #SBATCH --cpus-per-task={orch_cpus_per_task}
        #SBATCH --mem={orch_mem}
        #SBATCH --output={orch_output}
        {orch_time_line}

        module load OpenMPI/{orch_openmpi_ver}
        module load cuda/{orch_cuda_version}
        source ~/miniforge3/etc/profile.d/conda.sh
        conda activate {orch_conda_env}
        {_ld_lines}

        cd {work_dir}

        echo "Node: $(hostname)  Start: $(date)"
        python {out_py}
        echo "End: $(date)"
    """)

    os.makedirs(os.path.dirname(os.path.abspath(out_sh)), exist_ok=True)
    with open(out_sh, 'w') as fh:
        fh.write(content)
    os.chmod(out_sh, 0o755)
    return out_sh

def load_diffusivity_results(
    input_structures,
    n_h_values,
    results_root,
) -> pd.DataFrame:
    """Scan results dirs and return a summary DataFrame (T_K, D, sigma_D, R2, struct, n_h)."""

    records = []
    for struct_path in input_structures:
        struct_stem = os.path.splitext(os.path.basename(struct_path))[0]
        for n_h in n_h_values:
            run_name  = f'{struct_stem}_{n_h}H'
            table_txt = os.path.join(results_root, run_name, 'analysis', 'diffusivity_table.txt')
            if not os.path.exists(table_txt):
                print(f'  [MISSING] {table_txt}')
                continue
            df = pd.read_csv(
                table_txt,
                sep=r'\s+',
                comment='#',
                names=_COL_NAMES,
            )
            df['struct'] = struct_stem
            df['n_h']    = n_h
            records.append(df)

    if records:
        summary = pd.concat(records, ignore_index=True)
        print(f'Loaded {len(summary)} rows from {len(records)} completed runs.')
        _display(summary.head(12))
        return summary

def qc_minimizations(
    input_structures,
    n_h_values,
    results_root,
) -> pd.DataFrame:
    """Parse minimize_bare.out and minimize_h_{T}K.out; return QC DataFrame."""

    # Lazy import — parsers.py uses list[str] | None syntax (Python 3.10+)
    from models.parsers import parse_minimization_log

    min_records = []
    for struct_path in input_structures:
        struct_stem = os.path.splitext(os.path.basename(struct_path))[0]
        for n_h in n_h_values:
            run_name = f'{struct_stem}_{n_h}H'
            for label, logname in [('bare', 'minimize_bare.out'), ('H', 'minimize_h.out')]:
                logfile = os.path.join(results_root, run_name, 'lammps_scripts', logname)
                if not os.path.exists(logfile):
                    logfile = os.path.join(results_root, run_name, logname)
                if not os.path.exists(logfile):
                    continue
                _, meta = parse_minimization_log(logfile)
                min_records.append({
                    'struct':    struct_stem,
                    'n_h':       n_h,
                    'step':      label,
                    'Ecoh_eV':   meta.get('Ecoh_eV_per_atom', float('nan')),
                    'a0_Ang':    meta.get('a0_Angstrom',      float('nan')),
                    'Fmax':      meta.get('Fmax_eV_per_Ang',  float('nan')),
                    'converged': meta.get('stop_criterion',   'unknown'),
                })

    if not min_records:
        print('No minimisation logs found (jobs may still be running).')
        return pd.DataFrame()

    min_df = pd.DataFrame(min_records)
    _display(min_df)
    bad = min_df[~min_df['converged'].str.contains('force', case=False, na=False)]
    if not bad.empty:
        print(f'\n{len(bad)} minimisation(s) may not have converged:')
        _display(bad[['struct', 'n_h', 'step', 'Fmax', 'converged']])
    else:
        print('All minimisations converged on force criterion.')
    return min_df

def qc_nvt_thermo(
    input_structures,
    n_h_values,
    temperatures,
    results_root,
    timestep_ps: float = 0.0005,
) -> pd.DataFrame:
    """Parse NVT thermo logs; check temperature drift; save QC figures."""

    import matplotlib.pyplot as plt

    # Lazy import — parsers.py uses Python 3.10+ type union syntax
    from models.parsers import parse_equil_log, parse_thermo_series

    thermo_records = []
    for struct_path in input_structures:
        struct_stem = os.path.splitext(os.path.basename(struct_path))[0]
        for n_h in n_h_values:
            run_name = f'{struct_stem}_{n_h}H'
            for T in temperatures:
                out_file = os.path.join(
                    results_root, run_name, 'results', f'{T}K', f'nvt_{T}K.out'
                )
                meta = parse_equil_log(out_file)
                if meta:
                    thermo_records.append({
                        'struct':      struct_stem,
                        'n_h':         n_h,
                        'T_set':       T,
                        'T_final_K':   meta.get('temp_final_K', float('nan')),
                        'PE_final_eV': meta.get('pe_final_eV',  float('nan')),
                    })

    if thermo_records:
        thermo_df = pd.DataFrame(thermo_records)
        _display(thermo_df)
        thermo_df['T_drift_%'] = (
            (thermo_df['T_final_K'] - thermo_df['T_set']).abs()
            / thermo_df['T_set'] * 100
        )
        drifty = thermo_df[thermo_df['T_drift_%'] > 5]
        if not drifty.empty:
            print(f'\n{len(drifty)} run(s) with >5% temperature drift:')
            _display(drifty[['struct', 'n_h', 'T_set', 'T_final_K', 'T_drift_%']])
        else:
            print('All NVT runs within 5% of target temperature.')
    else:
        print('No NVT thermo logs found (jobs may still be running).')
        return pd.DataFrame()

    for struct_path in input_structures:
        struct_stem = os.path.splitext(os.path.basename(struct_path))[0]
        for n_h in n_h_values:
            run_name = f'{struct_stem}_{n_h}H'
            fig, axes = plt.subplots(
                2, len(temperatures),
                figsize=(3.5 * len(temperatures), 6),
                sharex='col',
            )
            has_data = False
            for col, T in enumerate(temperatures):
                out_file = os.path.join(
                    results_root, run_name, 'results', f'{T}K', f'nvt_{T}K.out'
                )
                msd_file = os.path.join(
                    results_root, run_name, 'results', f'{T}K', f'msd_{T}K.dat'
                )
                res = parse_thermo_series(out_file)
                if res is not None:
                    steps, temps_trace, _ = res
                    t_ps = steps * timestep_ps
                    axes[0][col].plot(t_ps, temps_trace, lw=0.6)
                    axes[0][col].axhline(T, color='red', lw=1, ls='--', label=f'{T} K')
                    axes[0][col].set_title(f'{T} K')
                    has_data = True
                if os.path.exists(msd_file):
                    try:
                        msd_data = np.loadtxt(msd_file, comments='#')
                        if msd_data.ndim == 2 and msd_data.shape[1] >= 2:
                            axes[1][col].plot(
                                msd_data[:, 0] * timestep_ps, msd_data[:, 1], lw=0.8
                            )
                            has_data = True
                    except Exception:
                        pass

            if has_data:
                axes[0][0].set_ylabel('Temperature (K)')
                axes[1][0].set_ylabel('MSD (Å²)')
                for col in range(len(temperatures)):
                    axes[1][col].set_xlabel('Time (ps)')
                fig.suptitle(f'{struct_stem} — {n_h} H atoms: Thermo QC', fontsize=12)
                fig.tight_layout()
                out_fig = os.path.join(
                    results_root, run_name, 'analysis', f'thermo_qc_{n_h}H.png'
                )
                os.makedirs(os.path.dirname(out_fig), exist_ok=True)
                plt.savefig(out_fig, dpi=120)
                plt.show()
                print(f'Saved: {out_fig}')
            else:
                plt.close()

    return thermo_df

def plot_D_vs_T(summary: pd.DataFrame, results_root: str) -> None:
    """D vs T plot (log scale, per structure, coloured by n_H)."""

    import matplotlib.pyplot as plt
    import matplotlib.cm as cm

    if summary.empty:
        print('No data to plot.')
        return

    structs  = summary['struct'].unique()
    n_h_list = sorted(summary['n_h'].unique())
    cmap      = cm.get_cmap('viridis', len(n_h_list))
    color_map = {nh: cmap(i) for i, nh in enumerate(n_h_list)}

    fig, axes = plt.subplots(
        1, len(structs),
        figsize=(5 * len(structs), 4),
        sharey=True,
        squeeze=False,
    )

    for col, struct in enumerate(structs):
        ax  = axes[0][col]
        sub = summary[summary['struct'] == struct]
        for n_h in n_h_list:
            d = sub[sub['n_h'] == n_h].sort_values('T_K')
            if d.empty:
                continue
            ax.errorbar(
                d['T_K'], d['D'],
                yerr=d['sigma_D'],
                fmt='o-', color=color_map[n_h],
                label=f'{n_h} H', capsize=3,
            )
        ax.set_xlabel('Temperature (K)')
        ax.set_title(struct)
        ax.set_yscale('log')
        ax.legend(title='n_H', fontsize=8)

    axes[0][0].set_ylabel('D (m²/s)')
    fig.suptitle('H Diffusivity vs Temperature', fontsize=13)
    fig.tight_layout()
    out_fig = os.path.join(results_root, 'D_vs_T.png')
    plt.savefig(out_fig, dpi=150)
    plt.show()
    print(f'Saved: {out_fig}')

def plot_Ea_vs_nH(
    input_structures,
    n_h_values,
    results_root,
) -> pd.DataFrame:
    """Arrhenius fits per (struct, n_h); plot Ea vs n_H; return arr_df."""

    import matplotlib.pyplot as plt
    from models.diffusivity_post_processing import run_arrhenius_pipeline

    arrhenius_records = []
    for struct_path in input_structures:
        struct_stem = os.path.splitext(os.path.basename(struct_path))[0]
        for n_h in n_h_values:
            run_name  = f'{struct_stem}_{n_h}H'
            table_txt = os.path.join(results_root, run_name, 'analysis', 'diffusivity_table.txt')
            if not os.path.exists(table_txt):
                continue
            arr = run_arrhenius_pipeline(
                diffusivity_file=table_txt,
                outdir=os.path.join(results_root, run_name, 'analysis'),
                plot_filename=f'arrhenius_{n_h}H.png',
            )
            arrhenius_records.append({
                'struct': struct_stem,
                'n_h':    n_h,
                'Ea':     arr['Ea'],
                'Ea_err': arr.get('Ea_err', 0.0),
                'D0':     arr['D0'],
                'D0_err': arr.get('D0_err', 0.0),
            })

    if not arrhenius_records:
        print('No Arrhenius fits computed.')
        return pd.DataFrame()

    arr_df = pd.DataFrame(arrhenius_records)
    _display(arr_df)

    fig, ax = plt.subplots(figsize=(6, 4))
    for struct, grp in arr_df.groupby('struct'):
        grp = grp.sort_values('n_h')
        ax.errorbar(
            grp['n_h'], grp['Ea'],
            yerr=grp['Ea_err'],
            fmt='o-', label=struct, capsize=4,
        )
    ax.set_xlabel('Number of H atoms')
    ax.set_ylabel('Activation energy Ea (eV)')
    ax.set_title('Arrhenius Ea vs H concentration')
    ax.legend()
    fig.tight_layout()
    out_fig = os.path.join(results_root, 'Ea_vs_nH.png')
    plt.savefig(out_fig, dpi=150)
    plt.show()
    print(f'Saved: {out_fig}')
    return arr_df

def plot_arrhenius_overlay(
    summary: pd.DataFrame,
    arr_df: pd.DataFrame,
    temperatures,
    n_h_values,
    results_root: str,
) -> None:
    """log D vs 1000/T Arrhenius overlay for all n_H."""

    import matplotlib.cm as cm
    from models.diffusivity_post_processing import arrhenius_D

    if arr_df is None or arr_df.empty:
        print('Run plot_Ea_vs_nH first to compute arr_df.')
        return

    T_fit     = np.linspace(min(temperatures) - 20, max(temperatures) + 20, 200)
    inv_T_fit = 1000.0 / T_fit

    structs   = arr_df['struct'].unique()
    n_h_sorted = sorted(n_h_values)
    cmap      = cm.get_cmap('viridis', len(n_h_sorted))
    color_map = {nh: cmap(i) for i, nh in enumerate(n_h_sorted)}

    fig, axes = plt.subplots(
        1, len(structs),
        figsize=(5 * len(structs), 4),
        sharey=True, squeeze=False,
    )

    for col, struct in enumerate(structs):
        ax      = axes[0][col]
        sub_sum = summary[summary['struct'] == struct]
        sub_arr = arr_df[arr_df['struct'] == struct]

        for _, row in sub_arr.iterrows():
            n_h   = int(row['n_h'])
            color = color_map[n_h]
